  import csv, sys, duckdb
  from datetime import datetime
  from fortuneteller import db, flows, study
  from fortuneteller.__main__ import follows_kinds
  
  store = sys.argv[1]
  scores = {}
  for r in csv.DictReader(open("docs/steps/exposure-scores.csv")):
      scores.setdefault(r["rubric"], {})[r["ticker"]] = int(r["score"])
  con = db.get_connection()
  raw = duckdb.connect(store, read_only=True)
  spy = [d for (d,) in raw.execute("select day from daily_bars where instrument='SPY / ES' order by day").fetchall()]
  for flow_name, kind, rubric in (("opec", study.OIL_UP, "oil"), ("cpi", study.ABOVE, "rates")):
      events, kinds, _ = follows_kinds(flows.flow_named(flow_name), con)
      e = min((e for e in events if e.event_ts >= datetime(2020, 1, 1) and kinds.get(e.event_id) == kind), key=lambda e: e.event_ts)
      first = study.first_reaction_day(e, study.STOCK)
      day = min(d for d in spy if d >= first)
      before = max(d for d in spy if d < day)
      def ret(t):
          rows = dict(raw.execute("select day, close from daily_bars where instrument=? and day in (?, ?)", [t, before, day]).fetchall())
          return rows[day] / rows[before] - 1 if len(rows) == 2 else None
      idx = ret("SPY / ES")
      hi = {t: ret(t) - idx for t, s in scores[rubric].items() if s >= 3 and ret(t) is not None}
      lo = {t: ret(t) - idx for t, s in scores[rubric].items() if s <= -3 and ret(t) is not None}
      sp = sum(hi.values()) / len(hi) - sum(lo.values()) / len(lo)
      print(f"{flow_name} {kind}: event {e.event_ts}, reaction close {day}, close before {before}, SPY {idx:+.4%}")
      print("  high:", ", ".join(f"{t} {v:+.2%}" for t, v in sorted(hi.items())))
      print("  low: ", ", ".join(f"{t} {v:+.2%}" for t, v in sorted(lo.items())))
      print(f"  spread by hand {sp:+.6f}")
