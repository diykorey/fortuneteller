  from datetime import datetime
  from fortuneteller import db, flows, study
  con = db.get_connection()
  def day_move(closes, reaction, unit):
      days = [c.day for c in closes]
      i = next(k for k, d in enumerate(days) if d >= reaction)
      return i, study.release_move(closes[i - 1], closes[i], unit)
  for flow_name, kind, market, cond in (("nfp", "above", "UST5Y / ZF", None), ("cpi", "above", "UST5Y / ZF", ("T5YIE", 63, 20.0))):
      flow = flows.flow_named(flow_name)
      kinds = study.release_kinds(flow, con=con)
      events = [e for e in flows.stored_events(flow, con=con) if e.event_ts >= datetime(2020, 1, 1) and kinds.get(e.event_id) == kind]
      series = study.MVP_PRICE_SERIES[market]; closes = study.stored_closes(market, con=con)
      kept = []
      for e in events:
          r = study.first_reaction_day(e, series)
          if cond:
              on, n, th = cond; cs = study.MVP_PRICE_SERIES[on]; cc = study.stored_closes(on, con=con)
              rr = study.first_reaction_day(e, cs); days = [c.day for c in cc]
              b = max(k for k, d in enumerate(days) if d < rr)
              mv = (cc[b].price - cc[b - n].price) * 100
              print(f"  {e.event_ts:%Y-%m-%d} {on} {n}-close move {mv:+.1f} bps -> {'kept' if mv > th else 'out'}")
              if not mv > th: continue
          kept.append(r)
      # the base: every day span of the market from the first kept reaction close on, as follows takes it
      spans = study.span_moves(closes, series.unit, 1)
      start = min(kept)
      base = study.base_of([s.move for k, s in spans.items() if k >= start])
      counts = {o: 0 for o in study.OUTCOMES}
      for r in kept:
          k = min(d for d in spans if d >= r)
          counts[study.outcome(spans[k].move, base.cut)] += 1
      print(flow_name, kind, market, "events", len(kept), counts, f"cut {base.cut:.3f}")
