"""Every cell of the page's ask against follows' line (step 32, step 2).
Usage: python3 -I cmp32.py data.json FOLLOWS_DIR  (FOLLOWS_DIR holds fol-FLOW.txt and fol-fomc-long.txt)"""
import json, re, sys, pathlib
data = json.load(open(sys.argv[1])); folder = pathlib.Path(sys.argv[2])
pat = re.compile(r"^(?P<kind>.+?)\s+(?P<span>day|week|month|quarter|year)\s+(?P<market>.+?)\s+(?P<n>\d+)\s+(?P<up>\d+)\s+(?P<same>\d+)\s+(?P<down>\d+)\s+(?P<out>\d+)/(?P<hole>\d+)\s+(?P<base>\S+)\s+(?P<answer>.+?)\s{2,}(?P<chance>\S+)\s+(?P<placebo>\S+)\s+(?:(?P<cycles>\d+: .+?)\s{2,})?(?P<graded>\d+): (?P<right>\d+) right, (?P<verdict>.+)$")
total = equal = 0; diffs = []; holds = []; periods = []
for f in data["ask"]:
    files = [folder / f"fol-{f['name']}.txt"] + ([folder / "fol-fomc-long.txt"] if f["name"] == "fomc" else [])
    lines = {}
    for path in files:
        for line in path.read_text().splitlines()[2:]:
            m = pat.match(line)
            if m:
                lines.setdefault((m["kind"].strip(), m["span"], m["market"].strip()), m)
    for c in f["cells"]:
        total += 1
        m = lines.get((c["kind"], c["span"], c["market"]))
        where = f"{f['name']} {c['kind']} {c['span']} {c['market']}"
        if m is None:
            diffs.append(f"{where}: no follows line"); continue
        want = {"up": int(m["up"]), "same": int(m["same"]), "down": int(m["down"]), "answer": m["answer"].strip(),
                "graded": int(m["graded"]), "right": int(m["right"]),
                "verdict": m["verdict"].replace("(the kind)", "").strip(),
                "kind_cell": m["verdict"].endswith("(the kind)"), "period": m["placebo"] == "period"}
        if m["cycles"]:
            n, verdict = m["cycles"].split(": ", 1)
            want["cycles"] = [int(n), verdict.strip()]
        got = {k: c[k] for k in want}
        bad = [k for k in want if want[k] != got[k]]
        b = m["base"].split("/"); pb = [c["base_up"], c["base_same"], c["base_down"]]
        if (pb[0] is None) != (len(b) == 1) or (pb[0] is not None and any(abs(x - float(y)) > 0.005 + 1e-9 for x, y in zip(pb, b))):
            bad.append(f"base {pb} vs {b}")
        if bad:
            diffs.append(f"{where}: {bad} page {got} follows {want}")
        else:
            equal += 1
        if c["verdict"] == "holds" and not c["kind_cell"] and not c["period"]:
            holds.append(f"{where}: {c['answer']}, {c['graded']} graded, {c['right']} right")
        if c["period"]:
            periods.append(f"{where}: {c['answer']}, verdict {c['verdict']}")
print(f"flows {len(data['ask'])}, cells {total}; equal to follows' line field by field (up, same, down, base within its printed 2 places, answer, graded, right, verdict, the kind's mark, the placebo's period, and the Fed's cycles on its long spans): {equal}")
for x in diffs: print("differs:", x)
print(f"holds, unmarked: {len(holds)}"); [print("  " + x) for x in holds]
print(f"period marks: {len(periods)}"); [print("  " + x) for x in periods]
