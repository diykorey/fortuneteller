The one-off check script (scratchpad/m40/check.py), as run:
  import json, sys
  from datetime import datetime
  from pathlib import Path
  from statistics import pstdev
  sys.path.insert(0, "scripts")
  import news_moves as nm
  from fortuneteller import db
  from fortuneteller.study import EXCESS_OVER, stored_closes
  from nasdaq_earnings import symbol
  from sec_8k_times import UNLISTED, MORE_FILERS
  sec = Path(sys.argv[1])
  rows, _ = nm.member_days(sec)
  bigs = sorted((r for r in rows if r[3]), key=lambda r: (r[0], r[1]))
  nofile = sorted((r for r in rows if not r[4]), key=lambda r: (r[0], r[1]))
  N = len(bigs); M = len(nofile)
  listing = json.loads((sec / "company_tickers.json").read_text())
  ciks = UNLISTED | {row["ticker"]: int(row["cik_str"]) for row in listing.values()}
  con = db.get_connection()
  idx = {c.day: c.price for c in stored_closes(EXCESS_OVER, con=con)}
  days = sorted(idx)
  def raw(day, ticker):
      st = [c for c in stored_closes(symbol(ticker), con=con) if c.day in idx]
      k = [c.day for c in st].index(day)
      ex = [st[i].price / st[i-1].price - idx[st[i].day] / idx[st[i-1].day] for i in range(k-504, k+1)]
      return ex[-1], pstdev(ex[:-1])
  def filed(day, ticker):
      out = []
      for cik in [ciks[ticker], *MORE_FILERS.get(ticker, [])]:
          p = sec / f"CIK{cik:010d}.json"
          if not p.exists(): continue
          ix = json.loads(p.read_text()); parts = [ix["filings"]["recent"]] + [json.loads((sec / f["name"]).read_text()) for f in ix["filings"]["files"]]
          for part in parts:
              for i, form in enumerate(part["form"]):
                  if form in ("8-K", "8-K/A"):
                      st = datetime.fromisoformat(part["acceptanceDateTime"][i].replace("Z", "+00:00"))
                      w = nm.window_day(st, days)
                      if w == day or abs((st.date() - day).days) <= 3:
                          out.append((form, cik, part["accessionNumber"][i], part["acceptanceDateTime"][i], part["items"][i], w == day))
      return out
  for label, lst, n in (("big", bigs, N), ("no 8-K", nofile, M)):
      for j in (0, n // 4, n // 2, 3 * n // 4, n - 1):
          day, ticker, move, isbig, items = lst[j]
          ex, sd = raw(day, ticker)
          print(f"{label} #{j+1} {day} {ticker} script {move:+.5f} hand {ex:+.5f} sd {sd:.5f} 3sd {3*sd:.5f} big {isbig} items {sorted(items)}")
          for f in filed(day, ticker): print("   ", *f)
