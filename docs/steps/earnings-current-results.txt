"""Step 33's reload on a copy: store the seed's earnings as load-releases does, then check every
stored report's id, detail and minute against the seed, computed apart from the flow."""
import csv
import sys
from datetime import UTC, datetime

import duckdb

from fortuneteller import flows

store, seed = sys.argv[1], sys.argv[2]
con = duckdb.connect(store)
before = con.execute(
    "select count(*) from event_instances where event_type = 'Earnings release'"
).fetchone()
batch = flows.EARNINGS_FLOW.load(flows.Path(seed), datetime.now(UTC))
flows.store_events(batch.events, con=con)
print("reload:", batch.report[0], "; stored before", before[0])

with open(seed, newline="") as handle:
    rows = list(csv.DictReader(handle))
by_day: dict[str, list[dict[str, str]]] = {}
for row in rows:
    by_day.setdefault(row["day"], []).append(row)
expected = {}
for day, reports in by_day.items():
    for n, row in enumerate(sorted(reports, key=lambda r: r["ticker"])):
        code = n * 10**10 + 34 * 10**8 + int(day.replace("-", ""))
        expected[code] = (f"{row['ticker']} {row['kind']}", f"{day} {row['time']}")
stored = {
    i: (d, ts.strftime("%Y-%m-%d %H:%M"))
    for i, d, ts in con.execute(
        "select event_id, detail, event_ts from event_instances "
        "where event_type = 'Earnings release'"
    ).fetchall()
}
equal = sum(1 for i, v in stored.items() if expected.get(i) == v)
print(f"ids: {equal} of {len(stored)} stored equal the seed's ({len(expected)} in the seed)")
for i in sorted(set(stored) ^ set(expected)):
    print("only in", "store" if i in stored else "seed", i, stored.get(i) or expected.get(i))
for i in sorted(set(stored) & set(expected)):
    if stored[i] != expected[i]:
        print("differs", i, stored[i], expected[i])
